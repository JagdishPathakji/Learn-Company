Complete Database Architecture

PK Primary Key FK Foreign Key CASCADE ON DELETE CASCADE SET NULL ON DELETE SET NULL

Lookup Tables

    PK
  • role_nameVARCHAR(50)
    UNIQUE NOT NULL

STATUS

  • status_idINT AUTO_INCREMENT
    PK
  • status_nameVARCHAR(50)
    UNIQUE NOT NULL

BLOCK_TYPE

  • block_type_idINT AUTO_INCREMENT
    PK
  • type_nameVARCHAR(50)
    UNIQUE NOT NULL

CATEGORY

  • category_idINT AUTO_INCREMENT
    PK
  • nameVARCHAR(100)
    UNIQUE NOT NULL

TAG

  • tag_idINT AUTO_INCREMENT
    PK
  • nameVARCHAR(50)
    UNIQUE NOT NULL

User Domain

USER

  • user_idBINARY(16)
    PK
  • usernameVARCHAR(100)
    UNIQUE NOT NULL
  • emailVARCHAR(255)
    UNIQUE NOT NULL
  • password_hashVARCHAR(255)
    NOT NULL
  • created_atTIMESTAMP

USER_ROLE

  • user_idBINARY(16)
    PK FK CASCADE USER(user_id)
  • role_idINT
    PK FK CASCADE ROLE(role_id)

Article Domain

ARTICLE

  • article_idBINARY(16)
    PK
  • author_idBINARY(16)
    FK USER(user_id)
  • current_published_version_idBINARY(16)
    FK SET NULL ARTICLE_VERSION(version_id)
  • created_atTIMESTAMP

ARTICLE_CATEGORY

  • article_idBINARY(16)
    PK FK CASCADE ARTICLE(article_id)
  • category_idINT
    PK FK CASCADE CATEGORY(category_id)

ARTICLE_TAG

  • article_idBINARY(16)
    PK FK CASCADE ARTICLE(article_id)
  • tag_idINT
    PK FK CASCADE TAG(tag_id)

Versioning & Content

ARTICLE_VERSION

  • version_idBINARY(16)
    PK
  • article_idBINARY(16)
    FK CASCADE ARTICLE(article_id)
  • version_numberINT
    NOT NULL
  • titleVARCHAR(255)
  • status_idINT
    FK STATUS(status_id)
  • created_byBINARY(16)
    FK USER(user_id)
  • created_atTIMESTAMP

CONTENT_BLOCK

  • block_idBINARY(16)
    PK
  • version_idBINARY(16)
    FK CASCADE ARTICLE_VERSION(version_id)
  • block_type_idINT
    FK BLOCK_TYPE(block_type_id)
  • sequence_orderINT
  • content_dataTEXT

Editorial Review

EDITORIAL_REVIEW

  • review_idBINARY(16)
    PK
  • version_idBINARY(16)
    FK CASCADE ARTICLE_VERSION(version_id)
  • editor_idBINARY(16)
    FK USER(user_id)
  • decisionENUM
    ('Approve', 'Reject', 'Needs Work')
  • feedbackTEXT
  • reviewed_atTIMESTAMP

User Engagement

USER_RATING

  • article_idBINARY(16)
    PK FK CASCADE ARTICLE(article_id)
  • user_idBINARY(16)
    PK FK CASCADE USER(user_id)
  • rating_valueDECIMAL(2,1)
    CHECK(rating_value 1 TO 5)
  • updated_atTIMESTAMP

USER_COMMENT

  • comment_idBINARY(16)
    PK
  • article_idBINARY(16)
    FK CASCADE ARTICLE(article_id)
  • user_idBINARY(16)
    FK CASCADE USER(user_id)
  • comment_textTEXT
    NOT NULL
  • created_atTIMESTAMP
